Esquema Relacional de Persistencia (Base de Datos MySQL)
Estructura de tablas, claves primarias (PK), foráneas (FK) y tipos de datos
usuarios Core Users
  • PK id_usuario INT AUTO_INC
  • nombre, apellido VARCHAR(100)
  • tipo_documento ENUM
  • numero_documento VARCHAR(20)
  • email VARCHAR(150)
  • password_hash VARCHAR(255)
  • FK id_rol INT
computadores Hardware Inventory
  • PK id_computador INT AUTO_INC
  • codigo_inventario VARCHAR(50)
  • numero_serie VARCHAR(100)
  • FK id_marca INT
  • procesador, ram_gb VARCHAR / INT
  • FK id_sala_actual INT
  • estado_activo ENUM
salas Laboratories
  • PK id_sala INT AUTO_INC
  • nombre_sala VARCHAR(100)
  • sede VARCHAR(100)
  • tiene_polo_a_tierra TINYINT(1)
  • tiene_estabilizador TINYINT(1)
  • tiene_red_structured TINYINT(1)
prestamos Transactions
  • PK id_prestamo INT AUTO_INC
  • FK id_computador INT
  • FK id_docente INT
  • FK id_auxiliar_entrega INT
  • fecha_prestamo DATETIME
  • estado_prestamo ENUM
traslados Transfers
  • PK id_traslado INT AUTO_INC
  • FK id_computador INT
  • FK id_sala_origen INT
  • FK id_sala_destino INT
  • FK id_usuario_autoriza INT
  • estado_traslado ENUM
bajas Decommission
  • PK id_baja INT AUTO_INC
  • FK id_computador INT
  • motivo_baja TEXT
  • dictamen_tecnico TEXT
  • fecha_baja DATETIME
